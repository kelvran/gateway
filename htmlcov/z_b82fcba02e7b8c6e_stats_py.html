1"""Statistical helpers for eval scoring. 

2 

3`PRD.md`'s success metric is explicit: never report a bare pass rate without 

4a confidence interval alongside it. The Wilson score interval (not a normal 

5approximation) is the right tool here because it stays well-behaved exactly 

6where a golden-tier eval set tends to live: small `n` and/or a proportion 

7near 0 or 1, both regimes where the normal approximation breaks down. 

8""" 

9 

10from __future__ import annotations 

11 

12import math 

13from statistics import NormalDist 

14from typing import NamedTuple 

15 

16# A closed interval's edge exactly touching 0.0 or 1.0 reads, to a human or 

17# a CI/CD gate, as "impossible" / "certain" — a claim no finite sample can 

18# actually support. The literal Wilson formula *does* produce an exact 1.0 

19# at successes == total (and an exact 0.0 at successes == 0): this clamp is 

20# a deliberate, documented engineering choice on top of the real formula, 

21# not a substitute for it, so the reported interval never overstates 

22# certainty a finite `n` can't back up. 

23_EPSILON = 1e-9 

24

26def wilson_interval( 

27 successes: int, total: int, confidence: float = 0.95 

28) -> tuple[float, float]: 

29 """Return the Wilson score confidence interval for a binomial proportion. 

30 

31 Args: 

32 successes: number of successful trials (0 <= successes <= total). 

33 total: number of trials. Must be > 0. 

34 confidence: confidence level in (0, 1), e.g. 0.95 for a 95% CI. 

35 

36 Returns: 

37 (lower, upper) bound tuple, each strictly within (0.0, 1.0). 

38 

39 Raises: 

40 ValueError: if `total` is not positive, `successes` is out of 

41 [0, total], or `confidence` is not in (0, 1). 

42 """ 

43 if total <= 0: 

44 raise ValueError("total must be > 0") 

45 if not 0 <= successes <= total: 

46 raise ValueError("successes must satisfy 0 <= successes <= total") 

47 if not 0 < confidence < 1: 

48 raise ValueError("confidence must be in the open interval (0, 1)") 

49 

50 z = NormalDist().inv_cdf(1 - (1 - confidence) / 2) 

51 n = float(total) 

52 p_hat = successes / n 

53 

54 denom = 1 + (z**2) / n 

55 centre = (p_hat + (z**2) / (2 * n)) / denom 

56 adjustment = (z / denom) * math.sqrt( 

57 (p_hat * (1 - p_hat)) / n + (z**2) / (4 * n**2) 

58 ) 

59 

60 lower = centre - adjustment 

61 upper = centre + adjustment 

62 

63 lower = min(max(lower, _EPSILON), 1 - _EPSILON) 

64 upper = min(max(upper, _EPSILON), 1 - _EPSILON) 

65 

66 return lower, upper

69def mixture_sprt_early_stop( 

70 successes: int, 

71 trials_run: int, 

72 baseline_pass_rate: float, 

73 confidence: float = 0.95, 

74 relative_mixing_variance: float = 1.0, 

75) -> bool: 

76 """Decide whether a group of repeated trials should stop now, via a 

77 real mixture sequential probability ratio test (mSPRT) — Robbins 

78 (1970), popularized for production A/B testing by Johari, Koomen, 

79 Pekelis & Walsh (KDD 2017), the same paper this project's own prior 

80 two-checkpoint design already cited as the authoritative source for 

81 why naively rechecking `wilson_interval` after every trial inflates 

82 the true false-positive rate (the classical optional-stopping / 

83 peeking failure mode). 

84 

85 Unlike that prior design, this is checked after *every single trial*, 

86 with no pre-declared checkpoints and no Bonferroni correction: by 

87 Ville's inequality, the mSPRT likelihood ratio is a nonnegative 

88 martingale under the null, so `P(exists n: stop) <= 1 - confidence` 

89 holds *simultaneously at every trial count* — the actual anytime-valid 

90 property the old design's checkpoint restriction was a conservative 

91 workaround for, not a full substitute. 

92 

93 The null-hypothesis variance `baseline_pass_rate * (1 - 

94 baseline_pass_rate)` is used FIXED, never re-estimated from this 

95 group's own running data — see docs/rfcs/2026-09-05-evals-mixture- 

96 sprt-early-stopping.md's "A deliberate choice" section: an 

97 online-estimated variance only gives an asymptotically-valid test, 

98 with real, measurable false-positive inflation at low trial counts 

99 unless a substantial warmup is spent first — a cost this early- 

100 stopping mechanism exists specifically to avoid paying. 

101 

102 Args: 

103 successes: successful trials so far in this group. 

104 trials_run: total trials so far in this group (>= successes). 0 

105 never stops (no data yet). 

106 baseline_pass_rate: the pass rate this group's running rate is 

107 tested against. Must be in the open interval (0, 1) — a 

108 baseline of exactly 0 or 1 gives zero null variance, a 

109 degenerate test this project has no reason to support. No 

110 default: this project has no production traffic yet to 

111 calibrate one — the operator must always supply it 

112 explicitly. 

113 confidence: confidence level in (0, 1). The false-positive rate is 

114 bounded by `1 - confidence`, at every trial count. 

115 relative_mixing_variance: the mSPRT's mixing-distribution 

116 variance, as a multiple of the null variance (Johari et al.'s 

117 own stated rule of thumb: "on the order of" it, hence a 

118 default of 1.0). Tunes detection speed/power only — provably 

119 never affects the false-positive-rate guarantee above, for 

120 any positive value. 

121 

122 Returns: 

123 True if the mSPRT likelihood ratio at this trial count has 

124 crossed `1 / (1 - confidence)` — a real, valid stop decision. 

125 False otherwise, including `trials_run == 0`. 

126 

127 Raises: 

128 ValueError: if `baseline_pass_rate`/`confidence` is not in the 

129 open interval (0, 1), `relative_mixing_variance` is not 

130 positive, or `successes` is out of `[0, trials_run]`. 

131 """ 

132 if not 0 < baseline_pass_rate < 1: 

133 raise ValueError("baseline_pass_rate must be in the open interval (0, 1)") 

134 if not 0 < confidence < 1: 

135 raise ValueError("confidence must be in the open interval (0, 1)") 

136 if relative_mixing_variance <= 0: 

137 raise ValueError("relative_mixing_variance must be > 0") 

138 if not 0 <= successes <= trials_run: 

139 raise ValueError("successes must satisfy 0 <= successes <= trials_run") 

140 

141 if trials_run == 0: 

142 return False 

143 

144 null_variance = baseline_pass_rate * (1 - baseline_pass_rate) 

145 mixing_variance = relative_mixing_variance * null_variance 

146 n = trials_run 

147 observed_effect = successes / n - baseline_pass_rate 

148 

149 n_times_mixing_variance = n * mixing_variance 

150 denominator = null_variance + n_times_mixing_variance 

151 # Computed in log-space, not as a raw likelihood ratio compared 

152 # against 1/alpha directly: an extreme deviation with enough trials 

153 # drives the real (non-log) ratio's exponent well past what 

154 # math.exp can represent (a real OverflowError, caught by this 

155 # module's own property-based test suite, not a hypothetical), while 

156 # the log-likelihood-ratio itself stays a perfectly ordinary, 

157 # representable float at every input this function accepts. 

158 log_likelihood_ratio = 0.5 * math.log(null_variance / denominator) + ( 

159 (n * n_times_mixing_variance * observed_effect**2) 

160 / (2 * null_variance * denominator) 

161 ) 

162 

163 alpha = 1 - confidence 

164 return log_likelihood_ratio >= -math.log(alpha)

167class ConfusionMatrix(NamedTuple): 

168 """A 2x2 confusion matrix for a binary judge verdict vs. a binary 

169 human/ground-truth label. `true_positive`/`false_positive` treat the 

170 judge's `True` (PASS) verdict as the positive class.""" 

171 

172 true_positive: int 

173 false_positive: int 

174 true_negative: int 

175 false_negative: int 

176 

177 

178def confusion_matrix( 

179 judge_verdicts: list[bool], human_verdicts: list[bool] 

180) -> ConfusionMatrix: 

181 """Return the 2x2 confusion matrix for paired judge/human verdicts. 

182 

183 Args: 

184 judge_verdicts: one bool per case, the judge's own PASS/FAIL 

185 verdict. 

186 human_verdicts: one bool per case, the ground-truth label for 

187 the same case, in the same order. Must be the same length 

188 as `judge_verdicts` and non-empty. 

189 

190 Returns: 

191 A `ConfusionMatrix` whose four counts sum to `len(judge_verdicts)`. 

192 

193 Raises: 

194 ValueError: if the two lists have different lengths, or both 

195 are empty. 

196 """ 

197 if len(judge_verdicts) != len(human_verdicts): 

198 raise ValueError( 

199 "judge_verdicts and human_verdicts must be the same length, " 

200 f"got {len(judge_verdicts)} and {len(human_verdicts)}" 

201 ) 

202 if len(judge_verdicts) == 0: 

203 raise ValueError("judge_verdicts/human_verdicts must be non-empty") 

204 

205 tp = fp = tn = fn = 0 

206 for judge, human in zip(judge_verdicts, human_verdicts, strict=True): 

207 if judge and human: 

208 tp += 1 

209 elif judge and not human: 

210 fp += 1 

211 elif not judge and not human: 

212 tn += 1 

213 else: 

214 fn += 1 

215 return ConfusionMatrix( 

216 true_positive=tp, false_positive=fp, true_negative=tn, false_negative=fn 

217 )

220def cohens_kappa(judge_verdicts: list[bool], human_verdicts: list[bool]) -> float: 

221 """Return Cohen's kappa — chance-corrected agreement between a 

222 judge's verdicts and human/ground-truth labels for the same cases. 

223 

224 Raw exact-match agreement systematically overstates how good a 

225 judge actually is: a real 2026 study found exact-match exceeds 

226 kappa by a ~38.6 percentage-point mean across 21 judges on MT-Bench 

227 — any judge-accuracy metric built only on raw exact-match will look 

228 far better than it actually is. Kappa corrects for this by 

229 subtracting out the agreement rate two independent, chance-only 

230 raters would produce given each one's own marginal PASS rate: 

231 

232 kappa = (p_o - p_e) / (1 - p_e) 

233 

234 where `p_o` is the observed agreement rate ((TP + TN) / N) and 

235 `p_e` is the chance-expected agreement rate derived from the 

236 judge's and human's own independent PASS rates. This is 

237 deliberately NOT the same as Pearson/Spearman/Kendall-tau_b/phi/ 

238 Matthews correlation coefficient — those five are mathematically 

239 identical to each other for binary verdicts, but kappa is a 

240 genuinely different, chance-discounted statistic, most different 

241 from them exactly when the judge's and human's marginal PASS rates 

242 diverge. 

243 

244 Args: 

245 judge_verdicts: one bool per case, the judge's own PASS/FAIL 

246 verdict. 

247 human_verdicts: one bool per case, the ground-truth label for 

248 the same case, in the same order. 

249 

250 Returns: 

251 Cohen's kappa, typically in [-1, 1] (1.0 = perfect agreement, 

252 0.0 = chance-level agreement, negative = worse than chance). 

253 

254 Raises: 

255 ValueError: on the same length-mismatch/empty conditions as 

256 `confusion_matrix`, or if `p_e == 1.0` — both raters have 

257 zero marginal variance in the identical direction (e.g. 

258 every verdict on both sides is `True`), making kappa 

259 undefined (a literal 0/0), not silently `NaN` or `1.0`. 

260 """ 

261 matrix = confusion_matrix(judge_verdicts, human_verdicts) 

262 n = float(len(judge_verdicts)) 

263 

264 p_observed = (matrix.true_positive + matrix.true_negative) / n 

265 judge_positive_rate = (matrix.true_positive + matrix.false_positive) / n 

266 human_positive_rate = (matrix.true_positive + matrix.false_negative) / n 

267 p_expected = judge_positive_rate * human_positive_rate + ( 

268 1 - judge_positive_rate 

269 ) * (1 - human_positive_rate) 

270 

271 if p_expected >= 1 - _EPSILON: 

272 raise ValueError( 

273 "cohens_kappa is undefined when p_expected == 1.0 " 

274 "(both raters have zero marginal variance in the same direction)" 

275 ) 

276 

277 return (p_observed - p_expected) / (1 - p_expected)
