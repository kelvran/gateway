1"""Score-cache key fabrication for LLM-judge calls. 

2 

3Deliberately a pure function with no I/O and no opinion on whether 

4caching is a good idea for a given case — that policy (opt-in via 

5`--use-score-cache`, built once from `--scores` before any new `Score` is 

6appended this run) lives in `evals.cli`, mirroring 

7`evals.rollout.cache.compute_run_cache_key`'s own exact precedent for 

8Run-level caching. 

9""" 

10 

11from __future__ import annotations 

12 

13import hashlib 

14import json 

15 

16 

17def compute_score_cache_key( 

18 output: str, 

19 reference: str, 

20 scorer_id: str, 

21 axis: str | None = None, 

22 debias: bool = False, 

23) -> str: 

24 """SHA-256 hex digest over the canonical JSON of 

25 `{output, reference, scorer_id, axis, debias}` — a judge verdict is a 

26 function of exactly these five inputs, plus the judge prompt template 

27 baked into `evals.judge.llm_judge.judge()` itself. Like 

28 `compute_run_cache_key`'s own deliberate omission of a harness/runner 

29 code version, this key does NOT version the prompt template — a 

30 prompt change is a code change, not a config change, and invalidating 

31 stale cached verdicts after one is the operator's responsibility 

32 (e.g. by pointing `--scores` at a fresh file), not solved with an 

33 unrequested prompt-hash field here. 

34 

35 `axis` (added 2026-09-05, per docs/rfcs/2026-09-05-evals-multi-axis- 

36 judging.md) is included in the key, but ONLY when given, so a cached 

37 "correctness" verdict for a given output can never be mistakenly 

38 served for a "safety" query against that same output — two genuinely 

39 different judgments, even though `output`/`reference`/`scorer_id` are 

40 identical. Omitting `axis` (the default, `None`) reproduces the exact 

41 original 3-field canonical JSON byte-for-byte, so every 

42 `score_cache_key` computed before this parameter existed still 

43 matches — a real backward-compatibility property, not just an 

44 unenforced intention, verified by 

45 `test_omitting_axis_matches_the_original_pre_axis_key_exactly`. 

46 

47 `debias` (added 2026-09-09, per docs/rfcs/2026-09-09-evals-judge- 

48 debiasing-position-swap.md) mirrors `axis`'s exact same pattern for 

49 the identical reason: a debiased (position-swapped, two-call) 

50 verdict and a non-debiased (single-call) verdict for the identical 

51 `(output, reference, scorer_id, axis)` are genuinely different 

52 judgments — a debiased-off run flipping `--judge-debias` on later 

53 must never reuse the other's cached verdict. Included in the key 

54 ONLY when `True` (the default `False` is omitted), so every 

55 `score_cache_key` computed before this parameter existed still 

56 matches byte-for-byte. 

57 

58 Canonicalization matches `compute_run_cache_key`'s own RFC-8785-style 

59 choices (`sort_keys`, no whitespace, ASCII-only, NaN/Infinity 

60 rejected) for the same stable-hashing correctness reasons. 

61 """ 

62 fields: dict[str, str | bool] = { 

63 "output": output, 

64 "reference": reference, 

65 "scorer_id": scorer_id, 

66 } 

67 if axis is not None: 

68 fields["axis"] = axis 

69 if debias: 

70 fields["debias"] = True 

71 canonical = json.dumps( 

72 fields, 

73 sort_keys=True, 

74 ensure_ascii=True, 

75 allow_nan=False, 

76 separators=(",", ":"), 

77 ) 

78 return hashlib.sha256(canonical.encode("utf-8")).hexdigest()
