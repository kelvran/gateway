1"""Self-contained OTel Python SDK span capture for real sandbox executions. 

2 

3No OTLP exporter is wired -- see docs/rfcs/2026-09-04-evals-trace-span- 

4model.md for why: the OTel SDK is used purely as a real, spec-compliant 

5trace_id/span_id + span-lifecycle generator (for possible future 

6correlation with gateway's own real OTel spans, per api/otel's still- 

7undecided transport), never as a live export pipeline. 

8 

9A locally-held `TracerProvider` is used rather than the process-wide 

10`trace.set_tracer_provider()` singleton -- the latter is `Once()`-guarded 

11and process-global (confirmed via source inspection of the installed SDK): 

12the first caller wins, every later call silently no-ops. That is wrong for 

13a library invoked repeatedly across CLI processes and, especially, across 

14an in-process pytest suite. `get_tracer()` is called directly on this 

15module's own `TracerProvider` instance instead. 

16 

17Attribute names are hardcoded string literals rather than imported from 

18`opentelemetry.semantic_conventions` (a 0.65b0 beta package) -- confirmed 

19via inspection that `process.command_args`/`process.exit.code` are real 

20but "_incubating"-only attributes (no stable `process` module exists in 

21this SDK version at all), while `container.image.name` is stable. 

22`gen_ai.operation.name` is deliberately never used here -- confirmed, via 

23the OTel semantic-conventions registry, to be a namespace for LLM/model- 

24inference operations only, not a container-sandbox execution. 

25""" 

26 

27from __future__ import annotations 

28 

29from opentelemetry.sdk.trace import ReadableSpan, SpanProcessor, TracerProvider 

30from opentelemetry.sdk.trace import Span as OTelSpan 

31from opentelemetry.trace import Status, StatusCode 

32 

33from evals.models import Span 

34 

35 

36class _CapturingSpanProcessor(SpanProcessor): 

37 """Captures the most recently ended span. 

38 

39 Safe only because `evals`' Rollout Scheduler starts and ends spans 

40 strictly sequentially -- never concurrently -- matching the scheduler's 

41 own existing single-worker design (see evals/ARCHITECTURE.md's Tech 

42 Stack: "asyncio at v1... concurrency is a later upgrade"). 

43 """ 

44 

45 def __init__(self) -> None: 

46 self.last_span: ReadableSpan | None = None 

47 

48 def on_end(self, span: ReadableSpan) -> None: 

49 self.last_span = span 

50 

51 

52_provider = TracerProvider() 

53_processor = _CapturingSpanProcessor() 

54_provider.add_span_processor(_processor) 

55_tracer = _provider.get_tracer("kelvran.evals") 

56 

57 

58def start_sandbox_span(*, image: str, command: list[str]) -> OTelSpan: 

59 """Start one real OTel span for a single `run_in_sandbox()` attempt. 

60 

61 Returns the live OTel `Span` (not `evals.models.Span`) -- the caller 

62 must eventually pass it to `finish_sandbox_span` to close it out and 

63 get the persisted pydantic model back. 

64 """ 

65 span = _tracer.start_span("sandbox.exec") 

66 span.set_attribute("process.command_args", command) 

67 span.set_attribute("container.image.name", image) 

68 return span 

69 

70 

71def finish_sandbox_span( 

72 otel_span: OTelSpan, 

73 *, 

74 run_id: str, 

75 image: str, 

76 command: list[str], 

77 exit_code: int | None, 

78 container_id: str | None, 

79 error: str | None, 

80) -> Span: 

81 """End `otel_span` and return the finished `evals.models.Span`. 

82 

83 Exactly one of `exit_code`/`error` is meaningful: `error` set means 

84 `run_in_sandbox()` itself raised (nothing ever ran, no exit code 

85 exists); otherwise `exit_code` is the real result. `container_id` is 

86 `None` only if the container never reached the point of being 

87 created -- never a fabricated placeholder. 

88 """ 

89 if error is not None: 

90 otel_span.set_status(Status(StatusCode.ERROR, description=error)) 

91 else: 

92 otel_span.set_attribute("process.exit.code", exit_code) 

93 otel_span.set_status(Status(StatusCode.OK)) 

94 if container_id is not None: 

95 otel_span.set_attribute("container.id", container_id) 

96 otel_span.end() 

97 

98 finished = _processor.last_span 

99 assert finished is not None, "on_end fires synchronously inside span.end()" # noqa: S101 -- internal OTel SDK invariant check, not user-facing validation 

100 ctx = finished.get_span_context() 

101 assert ctx is not None # noqa: S101 -- same internal invariant, not user-facing validation 

102 

103 return Span( 

104 span_id=format(ctx.span_id, "016x"), 

105 trace_id=format(ctx.trace_id, "032x"), 

106 parent_span_id=None, 

107 run_id=run_id, 

108 name=finished.name, 

109 start_time_unix_nano=finished.start_time, 

110 end_time_unix_nano=finished.end_time, 

111 status=finished.status.status_code.name, 

112 process_command_args=command, 

113 process_exit_code=exit_code, 

114 container_image_name=image, 

115 container_id=container_id, 

116 error=error, 

117 )
