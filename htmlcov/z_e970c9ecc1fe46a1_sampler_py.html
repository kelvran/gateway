1"""Sampling-rate and rule-filter decisions for a continuous stream of 

2`GatewayDecisionEvent`s — see this package's own doc comment for why 

3LLM-judge scoring of a sampled event is explicitly out of scope here. 

4 

5Deliberately a flat, third-layer module (per `.importlinter`'s `layers` 

6contract), sibling to `evals.auto_flag` on that same `|`-joined layer 

7bullet: imports only `evals.contracts.gatewayevents.v1` (a generated 

8protobuf module, not one of this project's own layered packages) and 

9stdlib `random` — never `evals.models`/`evals.ingestion.*`, since this 

10module operates on the RAW event, one stage before any `EvalCase`/`Run` 

11conversion or object-storage listing/decoding exists at all. The CLI 

12(`evals ingest`) is what wires this together with 

13`evals.ingestion.object_store`/`decode`. 

14"""

25def should_sample( 

26 event: GatewayDecisionEvent, 

27 sample_rate: float, 

28 rng: random.Random | None = None, 

29) -> bool: 

30 """Reports whether `event` is selected by a uniform-random sample at 

31 `sample_rate` — the same "configurable sampling rate for cost 

32 control" every real online-eval platform surveyed (Arize/LangSmith/ 

33 Braintrust) ships. `event` itself is currently unused (the decision 

34 is content-independent, a pure coin flip) but is a real parameter, 

35 not decoration — a future rate that varies by `event.outcome` (e.g. 

36 "always sample failures, 1% sample successes") slots in here without 

37 changing this function's own call sites, mirroring 

38 `evals.auto_flag.FlagRule.matches`'s identical "the event/case is 

39 always available to the decision, even when today's rule doesn't 

40 need every field" convention. 

41 

42 `sample_rate <= 0` always returns `False` (never sample) and 

43 `sample_rate >= 1` always returns `True` (always sample) — exact, 

44 deterministic boundaries, never left to `rng` to resolve by chance, 

45 so `sample_rate=1.0` (the default `evals ingest --sample-rate` 

46 ships, reproducing that command's pre-existing "ingest everything" 

47 behavior exactly) can never spuriously reject an event under any 

48 `rng` state. 

49 

50 `rng` defaults to a fresh, unseeded `random.Random()` per call when 

51 `None` — real production sampling has no reproducibility 

52 requirement. Callers doing anything deterministic/testable (this 

53 module's own test suite, or an operator's `--sample-seed` dry run) 

54 must pass an explicit, seeded `random.Random()` instance instead. 

55 """ 

56 if sample_rate <= 0: 

57 return False 

58 if sample_rate >= 1: 

59 return True 

60 active_rng = rng if rng is not None else random.Random() # noqa: S311 -- online-eval sampling, never cryptographic material 

61 return active_rng.random() < sample_rate

64@dataclass(frozen=True) 

65class RuleFilter: 

66 """One named rule matching a `GatewayDecisionEvent` on its real, 

67 already-populated fields — mirrors `evals.auto_flag.FlagRule`'s 

68 exact shape (name + a predicate), applied one stage earlier: against 

69 the raw event, before any `EvalCase`/`Run` conversion exists. 

70 """ 

71 

72 name: str 

73 matches: Callable[[GatewayDecisionEvent], bool] 

74 

75 

76DEFAULT_RULE_FILTERS: tuple[RuleFilter, ...] = ( 

77 RuleFilter( 

78 name="gateway-failure-outcome", 

79 matches=lambda event: ( 

80 event.outcome 

81 in ( 

82 GatewayDecisionEvent.Outcome.OUTCOME_UPSTREAM_ERROR, 

83 GatewayDecisionEvent.Outcome.OUTCOME_RATE_LIMITED, 

84 GatewayDecisionEvent.Outcome.OUTCOME_GUARDRAIL_BLOCKED, 

85 GatewayDecisionEvent.Outcome.OUTCOME_DEPLOYMENT_CAPACITY, 

86 ) 

87 ), 

88 ), 

89) 

90"""Mirrors `evals.auto_flag.DEFAULT_RULES`' own single outcome-based 

91rule and its exact 4-outcome set exactly — the two modules independently 

92converged on "failures are the interesting thing to flag" for their own 

93respective inputs (an ingested `EvalCase` there, a raw event here), so 

94this reuses that same, already-reasoned-about set rather than inventing 

95a second one. Kept as a SEPARATE tuple, not imported from `auto_flag`, 

96since the two operate on genuinely different input types 

97(`GatewayDecisionEvent` vs. `EvalCase.task_spec["outcome"]`'s string 

98form) — see `evals.auto_flag`'s own module doc comment for why 

99`Run.status` alone cannot distinguish these outcomes on already-ingested 

100data, the reason that module reads the string form at all; this module, 

101operating on the event before that lossy collapse, uses the real enum 

102values directly instead. 

103"""

106def passes_rule_filters( 

107 event: GatewayDecisionEvent, 

108 filters: tuple[RuleFilter, ...] = DEFAULT_RULE_FILTERS, 

109) -> bool: 

110 """Reports whether `event` matches AT LEAST ONE filter in `filters` 

111 — mirrors `evals.auto_flag.flag_candidates`'s own "matched by at 

112 least one rule" semantics. An EMPTY `filters` tuple means no rule 

113 constraint at all (every event passes) — an explicit, non-magic way 

114 for a caller to opt out of rule filtering entirely, matching 

115 `should_sample`'s own "sample_rate=1.0 means always" convention: 

116 `evals ingest --filter-rules` (the opt-in flag) passes 

117 `DEFAULT_RULE_FILTERS`; omitting it passes `()`. 

118 """ 

119 if not filters: 

120 return True 

121 return any(rule_filter.matches(event) for rule_filter in filters)
