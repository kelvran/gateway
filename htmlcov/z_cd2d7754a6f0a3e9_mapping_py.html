1"""Maps one decoded `GatewayDecisionEvent` into the `EvalCase`+`Run` pair 

2`evals promote` already reads from every other source (`evals rollout`'s 

3own results-store output), per docs/rfcs/2026-09-07-evals-trace-ingestion- 

4object-storage.md's own "Unresolved Questions" (now resolved, per 

5DECISIONS.md): live-sampled production data feeds `evals promote` exactly 

6like any other run source — no separate review path, no separate 

7persistence shape. 

8 

9The mapping is deliberately honest about what a `GatewayDecisionEvent` 

10does NOT carry. Per that RFC's own Context section: "No prompt/completion 

11content anywhere in this schema" — there is no real completion text for 

12`Run.stdout`, and no ground truth for `EvalCase.reference`; both are left 

13empty/`None` rather than fabricated. There is also no real per-request 

14wall-clock duration on the wire, so `Run.latency_ms` is `0.0` — not a 

15guess, the same "no genuine per-run timing exists" sentinel 

16`evals.rollout.scheduler.run_suite` already uses for a `status="skipped"` 

17or cache-hit `Run`, reused here rather than invented fresh. 

18 

19This is exactly why the synthetic `EvalCase.tier` is always 

20`"drift_sample"`: per docs/rfcs/2026-09-05-evals-golden-regression- 

21promotion.md's own "Hard-requiring --scores always" rejection reasoning, 

22`drift_sample` is the one tier whose promotion doesn't need a pass/fail 

23judge verdict — the one honest fit for output that was never captured. 

24Nothing here stops a later `evals promote --tier regression` against one 

25of these `Run`s (the "no separate review path" policy means ingested data 

26is never special-cased), but that command's own `--scores` failing-verdict 

27precondition can never be honestly satisfied for it, since no `Score` was 

28or could be produced for content that was never captured — a real 

29constraint of the data, not a gap in this mapping. 

30""" 

31 

32from __future__ import annotations 

33 

34from evals.contracts.gatewayevents.v1.gatewayevents_pb2 import GatewayDecisionEvent 

35from evals.models import EvalCase, Run 

36 

37 

38def gateway_decision_event_to_eval_case_and_run( 

39 event: GatewayDecisionEvent, 

40) -> tuple[EvalCase, Run]: 

41 """Build the `(EvalCase, Run)` pair for one decoded event. 

42 

43 `id`/`eval_case_id` are derived from `trace_id`+`span_id` — real, 

44 spec-compliant OTel identifiers unique to the request this event 

45 describes (per the proto's own field comments), never a random UUID 

46 that would sever the join back to the source trace. Revision is 

47 always `1`: each ingested event is a fresh, standalone case, never a 

48 revision of anything already registered. Every ingest of the same 

49 underlying object produces the same `(EvalCase.id, Run.id)` again — 

50 deliberately not deduplicated here (`results_store.py`'s append-only 

51 files carry no dedup for any other source either); a caller re- 

52 ingesting overlapping S3/GCS prefixes gets a real duplicate line, the 

53 same tradeoff already accepted everywhere else in this module. 

54 """ 

55 base_id = f"gatewayevents-{event.trace_id}-{event.span_id}" 

56 outcome_name = GatewayDecisionEvent.Outcome.Name(event.outcome) 

57 completed = event.outcome == GatewayDecisionEvent.Outcome.OUTCOME_OK 

58 

59 case = EvalCase( 

60 id=base_id, 

61 revision=1, 

62 task_spec={ 

63 "source": "gatewayevents_v1", 

64 "trace_id": event.trace_id, 

65 "span_id": event.span_id, 

66 "virtual_key_id": event.virtual_key_id, 

67 "requested_model": event.requested_model, 

68 "outcome": outcome_name, 

69 # occurred_at/rate_limit_fail_open/fallback_happened/ 

70 # fallback_from_deployment/budget_spent_usd were previously 

71 # decoded from the wire and then silently discarded here -- 

72 # `evals.auto_flag`'s own module docstring names this exact 

73 # gap as the reason a future rule can't use latency/cost/ 

74 # fallback signals against already-ingested data. 

75 # occurred_at via ToJsonString(), not the raw Timestamp 

76 # message, since task_spec is a plain dict that must stay 

77 # JSON-serializable end to end (results_store.py's append- 

78 # only JSONL persistence). 

79 "occurred_at": event.occurred_at.ToJsonString(), 

80 "rate_limit_fail_open": event.rate_limit_fail_open, 

81 "fallback_happened": event.fallback_happened, 

82 "fallback_from_deployment": event.fallback_from_deployment, 

83 # budget_spent_usd is already a decimal-as-string on the wire 

84 # (this project's own established convention for exact 

85 # decimal values crossing a serialization boundary) -- passed 

86 # through verbatim, never parsed into a float here. 

87 "budget_spent_usd": event.budget_spent_usd, 

88 }, 

89 reference=None, 

90 tier="drift_sample", 

91 tags=[f"gatewayevents-outcome:{outcome_name}"], 

92 ) 

93 

94 run = Run( 

95 id=base_id, 

96 eval_case_id=case.id, 

97 eval_case_revision=case.revision, 

98 harness_config={ 

99 "source": "gatewayevents_v1", 

100 "trace_id": event.trace_id, 

101 "span_id": event.span_id, 

102 }, 

103 status="completed" if completed else "error", 

104 latency_ms=0.0, 

105 stdout="", 

106 stderr=event.fallback_reason, 

107 error=None if completed else f"gateway outcome: {outcome_name}", 

108 ) 

109 return case, run
